// ══════════════════════════════════════════════════════════════════════════════ // PadelEye · GAS · rebuildRankingSessions() // ── ONE-SHOT: rebuilds Ranking_Sessions from scratch ────────────────────────── // // WHEN TO RUN: // Run once from the Apps Script editor (no deploy needed). // Safe to run multiple times — overwrites Ranking_Sessions each time. // // SOURCES: // Stats_Sessions → per-player match stats per session // Sessions_Sessions → session_mvp (authoritative MVP source) // // OUTPUT: // Ranking_Sessions → overwritten with correct aggregated totals // // COLUMN SCHEMA (Ranking_Sessions): // player | sessions | matches | wins | losses | // points_for | points_against | points_difference | // win_rate | mvp_count | updated_at // ══════════════════════════════════════════════════════════════════════════════ function rebuildRankingSessions() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const log = []; const now = new Date().toISOString(); // ── 1. Read Stats_Sessions ────────────────────────────────────────────────── const statsSheet = ss.getSheetByName('Stats_Sessions'); if (!statsSheet) { Browser.msgBox('Stats_Sessions not found'); return; } const statsRaw = statsSheet.getDataRange().getValues(); const sHdr = statsRaw[0].map(h => String(h).trim().toLowerCase()); // Column resolvers — accept both EN and PT field names const sCol = name => { const aliases = { player: ['player','jogador','player_name'], session_num: ['session_serial_number','sessao_numero','sessao','session'], matches: ['matches','partidas'], wins: ['wins','vitorias'], losses: ['losses','derrotas'], points_for: ['points_for','gw','pg','goals_for','pontos_favor'], points_against: ['points_against','gl','pc','goals_against','pontos_contra'], points_difference: ['points_difference','gd','saldo'], }; for (const a of (aliases[name] || [name])) { const idx = sHdr.indexOf(a); if (idx >= 0) return idx; } return -1; }; const S_PLAYER = sCol('player'); const S_SESSN = sCol('session_num'); const S_M = sCol('matches'); const S_W = sCol('wins'); const S_L = sCol('losses'); const S_PF = sCol('points_for'); const S_PA = sCol('points_against'); if (S_PLAYER < 0) { Browser.msgBox('Cannot find player column in Stats_Sessions'); return; } // Aggregate per player const playerMap = {}; // { name: { sessions: Set, matches, wins, losses, pf, pa } } for (let i = 1; i < statsRaw.length; i++) { const row = statsRaw[i]; const player = String(row[S_PLAYER] || '').trim(); if (!player) continue; if (!playerMap[player]) { playerMap[player] = { sessions: new Set(), matches: 0, wins: 0, losses: 0, pf: 0, pa: 0 }; } const p = playerMap[player]; // Track unique sessions via serial number const sn = String(S_SESSN >= 0 ? (row[S_SESSN] || '') : '').replace(/\D/g, '').trim(); if (sn) p.sessions.add(sn); p.matches += _num(row[S_M]); p.wins += _num(row[S_W]); p.losses += _num(row[S_L]); p.pf += _num(row[S_PF]); // points_against: try direct column, else derive from pf - pd if (S_PA >= 0) { p.pa += _num(row[S_PA]); } else { const pdCol = sCol('points_difference'); if (pdCol >= 0) p.pa += (_num(row[S_PF]) - _num(row[pdCol])); } } log.push('Stats_Sessions rows read: ' + (statsRaw.length - 1)); log.push('Unique players found: ' + Object.keys(playerMap).length); // ── 2. Read Sessions_Sessions → authoritative MVP source ──────────────────── const sessSheet = ss.getSheetByName('Sessions_Sessions'); const mvpMap = {}; // { player_name: count } if (sessSheet) { const sessRaw = sessSheet.getDataRange().getValues(); const sesHdr = sessRaw[0].map(h => String(h).trim().toLowerCase()); const mvpCol = _findCol(sesHdr, ['session_mvp','mvp','session_mvp_name']); const numCol = _findCol(sesHdr, ['session_serial_number','numero','session_number']); if (mvpCol >= 0) { for (let i = 1; i < sessRaw.length; i++) { const mvpName = String(sessRaw[i][mvpCol] || '').trim(); if (mvpName && mvpName !== '—' && mvpName !== '') { mvpMap[mvpName] = (mvpMap[mvpName] || 0) + 1; // Also supplement session counts if Stats_Sessions was missing rows if (playerMap[mvpName] && numCol >= 0) { const sn = String(sessRaw[i][numCol] || '').replace(/\D/g,'').trim(); if (sn) playerMap[mvpName].sessions.add(sn); } } } } log.push('Sessions_Sessions rows read: ' + (sessRaw.length - 1)); log.push('MVP entries found: ' + Object.keys(mvpMap).map(k => k + '×' + mvpMap[k]).join(', ')); } else { log.push('⚠️ Sessions_Sessions not found — MVP counts will be 0'); } // ── 3. Build ranking rows ─────────────────────────────────────────────────── const rankingRows = Object.entries(playerMap).map(([player, d]) => { const sessions = d.sessions.size; const matches = d.matches; const wins = d.wins; const losses = d.losses; const pf = d.pf; const pa = d.pa; const pd = pf - pa; const wr = matches > 0 ? Math.round(wins / matches * 100) + '%' : '0%'; const mvpCount = mvpMap[player] || 0; return [player, sessions, matches, wins, losses, pf, pa, pd, wr, mvpCount, now]; }); // Sort: WR desc, then wins desc, then sessions desc rankingRows.sort((a, b) => { const wrA = parseInt(a[8]), wrB = parseInt(b[8]); if (wrB !== wrA) return wrB - wrA; if (b[3] !== a[3]) return b[3] - a[3]; // wins return b[1] - a[1]; // sessions }); log.push('Ranking rows to write: ' + rankingRows.length); // ── 4. Write to Ranking_Sessions ──────────────────────────────────────────── const rankSheet = ss.getSheetByName('Ranking_Sessions'); if (!rankSheet) { Browser.msgBox('Ranking_Sessions sheet not found'); return; } // Clear all data rows (keep header row 1) const lastRow = rankSheet.getLastRow(); if (lastRow > 1) { rankSheet.getRange(2, 1, lastRow - 1, rankSheet.getLastColumn()).clearContent(); } // Ensure header is correct const HEADER = ['player','sessions','matches','wins','losses', 'points_for','points_against','points_difference', 'win_rate','mvp_count','updated_at']; rankSheet.getRange(1, 1, 1, HEADER.length).setValues([HEADER]); // Write data if (rankingRows.length > 0) { rankSheet.getRange(2, 1, rankingRows.length, HEADER.length).setValues(rankingRows); } // ── 5. Report ────────────────────────────────────────────────────────────── const report = [ '✅ Ranking_Sessions rebuilt successfully', '', 'Players: ' + rankingRows.length, '', 'Top 5:', ...rankingRows.slice(0, 5).map((r, i) => ` ${i+1}. ${r[0]} — ${r[8]} WR · ${r[3]}V · ${r[9]} MVPs` ), '', 'Details:', ...log, '', 'Timestamp: ' + now, ].join('\n'); Logger.log(report); Browser.msgBox(report); } // ── Helpers ─────────────────────────────────────────────────────────────────── function _num(v) { const n = parseFloat(String(v || '0').replace(/[^0-9.\-]/g, '')); return isNaN(n) ? 0 : n; } function _findCol(hdr, aliases) { for (const a of aliases) { const idx = hdr.indexOf(a.toLowerCase()); if (idx >= 0) return idx; } return -1; } // ══════════════════════════════════════════════════════════════════════════════ // VERIFY — run after rebuild to spot-check the output // ══════════════════════════════════════════════════════════════════════════════ function verifyRanking() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const rankSheet = ss.getSheetByName('Ranking_Sessions'); if (!rankSheet) { Browser.msgBox('Ranking_Sessions not found'); return; } const data = rankSheet.getDataRange().getValues(); const hdr = data[0]; const rows = data.slice(1).filter(r => r[0]); const playerCol = hdr.indexOf('player'); const wrCol = hdr.indexOf('win_rate'); const mvpCol = hdr.indexOf('mvp_count'); const sessCol = hdr.indexOf('sessions'); const winsCol = hdr.indexOf('wins'); const report = [ `Ranking_Sessions — ${rows.length} players`, '', '# | Player | Sessions | Wins | WR | MVPs', ...rows.map((r, i) => `${i+1}. ${r[playerCol]} | ${r[sessCol]}S | ${r[winsCol]}V | ${r[wrCol]} | 🏅${r[mvpCol]}` ), ].join('\n'); Logger.log(report); Browser.msgBox(report); }
